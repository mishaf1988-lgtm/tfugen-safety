<script>
(function(){
  var ROOMS=['חדר המחקר','חדר המסמכים','חדר הדוחות','תזכורות'];
  var ROOM_SUB={'חדר המחקר':'לומדים ומחפשים בחוץ','חדר המסמכים':'מייל, תיקיות ותוקף','חדר הדוחות':'דוחות, ביקורות ומגמות','תזכורות':'בדיקות חד-פעמיות'};
  var LABEL={done:'סיים',run:'עובד',fail:'נכשל',wait:'מחכה לך',never:'עוד לא רץ'};
  var agents=[],meta=null,open=null;
  function el(tag,cls,txt){var e=document.createElement(tag);if(cls)e.className=cls;if(txt!=null)e.textContent=txt;return e;}
  // DD/MM/YYYY HH:MM, Israel time
  function when(iso){
    if(!iso)return '';
    var d=new Date(iso);if(isNaN(d))return '';
    var p={};new Intl.DateTimeFormat('en-GB',{timeZone:'Asia/Jerusalem',day:'2-digit',month:'2-digit',year:'numeric',hour:'2-digit',minute:'2-digit',hour12:false})
      .formatToParts(d).forEach(function(x){p[x.type]=x.value;});
    return p.day+'/'+p.month+'/'+p.year+' '+p.hour+':'+p.minute;
  }
  function state(a){
    if(a.waiting)return 'wait';
    return ({done:1,run:1,fail:1,never:1})[a.status]?a.status:'never';
  }
  function render(){
    var floor=document.getElementById('floor');
    floor.textContent='';
    var stamp=document.getElementById('stamp');
    stamp.textContent=meta&&meta.updated_at?('עודכן '+when(meta.updated_at)):(agents.length?'':'עוד לא עודכן');
    var strip=document.getElementById('strip');strip.textContent='';
    var cnt={done:0,run:0,fail:0,wait:0,never:0};
    agents.forEach(function(a){cnt[state(a)]++;});
    var total=el('span','chip');total.appendChild(el('b',null,String(agents.length)));total.appendChild(document.createTextNode(' סוכנים'));strip.appendChild(total);
    ['run','wait','fail','done','never'].forEach(function(k){
      if(!cnt[k])return;
      var c=el('span','chip s-'+k);c.appendChild(el('i','dot'));c.appendChild(el('b',null,String(cnt[k])));c.appendChild(document.createTextNode(' '+LABEL[k]));strip.appendChild(c);
    });
    var inbox=document.getElementById('inbox'),list=document.getElementById('inbox-list');list.textContent='';
    var waiting=agents.filter(function(a){return a.waiting;});
    inbox.hidden=!waiting.length;
    waiting.forEach(function(a){var li=el('li');li.appendChild(el('strong',null,a.name+': '));li.appendChild(el('span',null,a.waiting));list.appendChild(li);});
    if(!agents.length){floor.appendChild(el('div','empty','המשרד ריק עדיין. כל בוקר ב-07:38 סוכן קטן עובר על כל הסוכנים ושם כל אחד בשולחן שלו.'));return;}
    var rooms=ROOMS.slice();
    agents.forEach(function(a){if(a.room&&rooms.indexOf(a.room)<0)rooms.push(a.room);});
    rooms.forEach(function(r){
      var inRoom=agents.filter(function(a){return (a.room||'חדר המסמכים')===r;});
      if(!inRoom.length)return;
      inRoom.sort(function(x,y){return (x.order||99)-(y.order||99)||String(x.name).localeCompare(String(y.name),'he');});
      var room=el('section','room');room.appendChild(el('h2',null,r));
      if(ROOM_SUB[r])room.appendChild(el('div','sub',ROOM_SUB[r]));
      var desks=el('div','desks');
      inRoom.forEach(function(a){
        var s=state(a);
        var b=el('button','desk s-'+s);b.type='button';b.id='desk-'+a.id;
        b.setAttribute('aria-expanded',open===a.id?'true':'false');
        var top=el('div','top');top.appendChild(el('i','dot'));top.appendChild(el('span','name',a.name));b.appendChild(top);
        if(a.role)b.appendChild(el('div','role',a.role));
        b.appendChild(el('div','when',LABEL[s]+(a.last_run?', '+when(a.last_run):'')));
        b.addEventListener('click',function(){open=(open===a.id)?null:a.id;render();});
        desks.appendChild(b);
        if(open===a.id){
          var d=el('div','detail');var dl=el('dl');
          [['מתי',a.schedule],['ריצה אחרונה',a.last_run?when(a.last_run)+' ('+LABEL[a.status]+')':'עוד לא רץ'],['ריצה הבאה',when(a.next_run)],
           ['מה עשה',a.did],['מה למד',a.learned],['מחכה לך',a.waiting],['עלות לריצה',a.cost]].forEach(function(r2){
            if(!r2[1])return;dl.appendChild(el('dt',null,r2[0]));dl.appendChild(el('dd',null,r2[1]));
          });
          d.appendChild(dl);
          // 17/10/2026 (Michael: "all my agents must work from the project"): the Routine's own page
          // in claude.ai lists every run in full and has Run now. The desk id is the trigger id.
          if(/^trig_[A-Za-z0-9]+$/.test(a.id)){
            var go=el('a','go','\u05e4\u05ea\u05d7 \u05d0\u05ea \u05d4\u05e1\u05d5\u05db\u05df');go.href='https://claude.ai/code/routines/'+a.id;go.target='_blank';go.rel='noopener';
            d.appendChild(go);
            d.appendChild(el('div','hint','\u05e9\u05dd: \u05db\u05dc \u05d4\u05e8\u05d9\u05e6\u05d5\u05ea \u05d1\u05de\u05dc\u05d5\u05d0\u05df, \u05d5\u05db\u05e4\u05ea\u05d5\u05e8 Run now \u05dc\u05d4\u05e4\u05e2\u05dc\u05d4 \u05d1\u05d9\u05d3.'));
          }
          desks.appendChild(d);
        }
      });
      room.appendChild(desks);floor.appendChild(room);
    });
  }
  render();
  if(!window.claude||!window.claude.use){document.getElementById('stamp').textContent='אין חיבור לנתונים בתצוגה הזו';return;}
  window.claude.use('db').then(function(db){
    if(!db){document.getElementById('stamp').textContent='צריך להיכנס לחשבון כדי לראות את המשרד';return;}
    db.collection('agents').onSnapshot(function(snap){
      agents=snap.docs.map(function(d){var x=d.data()||{};x.id=d.id;return x;});render();
    },function(){document.getElementById('stamp').textContent='לא הצלחתי לקרוא את המשרד, נסה לרענן';});
    db.doc('meta/office').onSnapshot(function(s){meta=s.data()||null;render();},function(){});
  });
})();
</script>
