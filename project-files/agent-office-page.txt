  // All desks on one screen: try every column count and keep the one with the largest desks.
  function fit(){
    var d=document.querySelector('.office .desks');if(!d)return;
    var n=d.children.length;if(!n)return;
    var W=d.clientWidth-12,H=d.clientHeight-12,G=6,best=1,bs=-1;
    for(var c=1;c<=n;c++){
      var r=Math.ceil(n/c),cw=(W-(c-1)*G)/c,ch=(H-(r-1)*G)/r;
      var sc=Math.min(cw/160,ch/200);
      if(sc>bs){bs=sc;best=c;}
    }
    d.style.gridTemplateColumns='repeat('+best+',minmax(0,1fr))';
    // A list longer than its desk shows what fits and "ועוד N"; the click opens all of them.
    Array.prototype.forEach.call(d.querySelectorAll('.tasks'),function(ul){
      var more=ul.querySelector('.more');if(more)ul.removeChild(more);
      var li=Array.prototype.slice.call(ul.children);li.forEach(function(x){x.style.display='';});
      if(ul.scrollHeight<=ul.clientHeight+1)return;
      more=el('li','more');ul.appendChild(more);
      for(var k=li.length-1;k>0&&ul.scrollHeight>ul.clientHeight+1;k--){li[k].style.display='none';more.textContent='ועוד '+(li.length-k);}
    });
  }

  window.addEventListener('resize',fit);

    // 18/10/2026: the worker -> session map lives in meta/workers, so the weekly rotation (routine-rotation.md)
    // changes it without a new page version.
    db.doc('meta/workers').onSnapshot(function(s){var m=s.data();if(m&&m.sessions){Object.keys(m.sessions).forEach(function(k){SESS[k]=m.sessions[k];});render();}},function(){});
    db.doc('meta/office').onSnapshot(function(s){meta=s.data()||null;render();},function(){});
  });
